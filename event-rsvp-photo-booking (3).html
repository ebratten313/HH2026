const SHEET_NAME = 'RSVPs'; const HEADERS = ['Submitted','Family','Contact','Email','Mobile','Adults','Children', 'Children names/ages','Notes','Photos','Photo time','# photographed','Requests']; function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); sheet.appendRow(HEADERS); } return sheet; } function doGet(e) { const action = e.parameter.action; const sheet = getSheet_(); const rows = sheet.getDataRange().getValues().slice(1); if (action === 'config') { const raw = PropertiesService.getScriptProperties().getProperty('eventConfig'); return jsonOut_({ config: raw ? JSON.parse(raw) : null }); } if (action === 'slots') { const booked = rows.filter(r => r[9] === 'Yes' && r[10]).map(r => r[10]); return jsonOut_({ booked: booked }); } if (action === 'stats') { let people = 0, photos = 0; rows.forEach(r => { people += (Number(r[5]) || 0) + (Number(r[6]) || 0); if (r[9] === 'Yes') photos++; }); return jsonOut_({ families: rows.length, people: people, photos: photos }); } return jsonOut_({ error: 'unknown action' }); } function doPost(e) { const lock = LockService.getScriptLock(); try { lock.waitLock(10000); const body = JSON.parse(e.postData.contents); if (body.action === 'saveConfig') { PropertiesService.getScriptProperties().setProperty('eventConfig', JSON.stringify(body.config)); return jsonOut_({ status: 'ok' }); } const sheet = getSheet_(); if (body.slot) { const rows = sheet.getDataRange().getValues().slice(1); const taken = rows.some(r => r[9] === 'Yes' && r[10] === body.slot); if (taken) return jsonOut_({ status: 'taken' }); } sheet.appendRow([ body.submitted || new Date().toISOString(), body.family || '', body.contact || '', body.email || '', body.phone || '', body.adults || '', body.kids || '', body.kidnames || '', body.notes || '', body.photos || '', body.slot || '', body.photocount || '', body.photorequests || '' ]); return jsonOut_({ status: 'ok' }); } catch (err) { return jsonOut_({ status: 'error', message: String(err) }); } finally { lock.releaseLock(); } } function jsonOut_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
